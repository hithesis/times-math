\newcommand\checkintegral[2]{%
  \setbox0=\hbox{$\m@th#1\int$}%
  \setbox2=\hbox{$\m@th#1#2$}%
  \dimen0=\dimexpr\ht2+\dp2-\ht0-\dp0\relax
  \dimen2=.02pt
  \ifx#2\upint \dimen2=.15\dimexpr\ht0+\dp0\relax\fi
  \ifx#2\lowint \dimen2=.15\dimexpr\ht0+\dp0\relax\fi
  \ifdim\dimen0<-.02pt
    \PackageError{test-integrals}{Integral curve is too short for \string#2}{}%
  \fi
  \ifdim\dimen0>\dimen2
    \PackageError{test-integrals}{Height mismatch for \string#2}{}%
  \fi
  \dimen0=\dimexpr\ht0-\dp0-\ht2+\dp2\relax
  \ifdim\dimen0<0pt \dimen0=-\dimen0\fi
  \ifdim\dimen0>\dimen2
    \PackageError{test-integrals}{Axis mismatch for \string#2}{}%
  \fi}
\makeatother
